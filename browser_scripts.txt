Get all RDS user DAta

fetch(window.HAPPNIX_RUNTIME_CONFIG.buildApiUrl('/api/auth'), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ actionItem: 'GetDevAllUsers' })
}).then(res => res.json()).then(data => {
  console.log("Here are all the users from RDS:");
  console.table(data.users); // This will print them in a nice formatted table!
});

GET DynamoDB table names (needed for HomePage API)

fetch(window.HAPPNIX_RUNTIME_CONFIG.buildApiUrl('/api/home'), {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ actionItem: 'GetDevTableNames' })
}).then(r => r.json()).then(d => {
  console.log("DynamoDB table names:", d);
  window.HAPPNIX_RUNTIME_CONFIG.tables = d;
});
